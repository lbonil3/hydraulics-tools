```js
(function (root) {
  function resolvePrice(row) {
    const hn = Number(row.houston_n) || 0;
    const sn = Number(row.statewide_n) || 0;
    if (hn > 0 && row.houston_avg != null && !Number.isNaN(Number(row.houston_avg))) {
      return { price: Number(row.houston_avg), n: hn, source: 'Houston' };
    }
    if (sn > 0 && row.statewide_avg != null && !Number.isNaN(Number(row.statewide_avg))) {
      return { price: Number(row.statewide_avg), n: sn, source: 'Statewide' };
    }
    return { price: null, n: null, source: null };
  }

  function filterRows(rows, query) {
    const q = String(query || '').trim().toLowerCase();
    if (!q) return rows.slice();
    return rows.filter(function (r) {
      const code = String(r.bid_code || '').toLowerCase();
      const desc = String(r.description || '').toLowerCase();
      return code.indexOf(q) !== -1 || desc.indexOf(q) !== -1;
    });
  }

  function formatMoney(n) {
    if (n == null || Number.isNaN(Number(n))) return '—';
    return '$' + Number(n).toLocaleString('en-US', {
      minimumFractionDigits: 2,
      maximumFractionDigits: 2
    });
  }

  const api = { resolvePrice: resolvePrice, filterRows: filterRows, formatMoney: formatMoney };
  if (typeof module !== 'undefined' && module.exports) module.exports = api;
  root.BidCostLookup = api;
})(typeof globalThis !== 'undefined' ? globalThis : this);
```

```js
#!/usr/bin/env node
const fs = require('fs');
const path = require('path');
const dir = __dirname;
const dataDir = path.join(dir, 'data');

function parseCsv(text) {
  const lines = text.trim().split(/\r?\n/);
  const headers = lines[0].split(',');
  return lines.slice(1).map(function (line) {
    const cols = line.split(',');
    const o = {};
    headers.forEach(function (h, i) { o[h] = cols[i]; });
    return o;
  });
}

function loadSide(file, avgKey, nKey) {
  const rows = parseCsv(fs.readFileSync(path.join(dataDir, file), 'utf8'));
  const map = Object.create(null);
  rows.forEach(function (r) {
    const code = r.bid_code;
    if (!code) return;
    map[code] = {
      bid_code: code,
      description: r.description || '',
      unit: r.unit || '',
      avg: r.avg_unit_price === '' || r.avg_unit_price == null ? null : Number(r.avg_unit_price),
      n: Number(r.n_observations) || 0
    };
  });
  return map;
}

const houston = loadSide('bid-unit-prices-houston.csv');
const statewide = loadSide('bid-unit-prices-statewide.csv');
const metaIn = JSON.parse(fs.readFileSync(path.join(dataDir, '_meta.json'), 'utf8'));
const codes = Object.keys(Object.assign({}, statewide, houston)).sort();
const outRows = [];
codes.forEach(function (code) {
  const h = houston[code];
  const s = statewide[code];
  const houston_n = h ? h.n : 0;
  const statewide_n = s ? s.n : 0;
  if (houston_n <= 0 && statewide_n <= 0) return;
  outRows.push({
    bid_code: code,
    description: (h && h.description) || (s && s.description) || '',
    unit: (h && h.unit) || (s && s.unit) || '',
    houston_avg: h && houston_n > 0 ? h.avg : null,
    houston_n: houston_n,
    statewide_avg: s && statewide_n > 0 ? s.avg : null,
    statewide_n: statewide_n
  });
});

const out = {
  meta: {
    accessed: metaIn.access_date || '',
    let_date_min: metaIn.min_date || '',
    let_date_max: metaIn.max_date || '',
    source_url: 'https://data.texas.gov/dataset/Bid-Tabulations/de7b-7dna',
    note: 'Low bidder (rank=1) averages; planning-level only'
  },
  rows: outRows
};
fs.writeFileSync(path.join(dir, 'prices.json'), JSON.stringify(out));
console.log('wrote', outRows.length, 'rows');
```

- [ ] **Step 2: Create** `bid-cost/app.js`:

```js
(function () {
  const L = window.BidCostLookup;
  const tbody = document.getElementById('tbody');
  const search = document.getElementById('search');
  const count = document.getElementById('count');
  const metaLine = document.getElementById('metaLine');
  let allRows = [];
  const PAGE = 200; // initial paint + grow on scroll if needed; or render filtered capped

  function render(rows) {
    const resolved = rows.map(function (r) {
      const x = L.resolvePrice(r);
      return { r: r, x: x };
    }).filter(function (o) { return o.x.price != null; });

    count.textContent = resolved.length + ' item' + (resolved.length === 1 ? '' : 's');
    const slice = resolved.slice(0, PAGE);
    const html = slice.map(function (o) {
      return '<tr>' +
        '<td>' + escapeHtml(o.r.bid_code) + '</td>' +
        '<td>' + escapeHtml(o.r.description) + '</td>' +
        '<td>' + escapeHtml(o.r.unit) + '</td>' +
        '<td>' + L.formatMoney(o.x.price) + '</td>' +
        '<td>' + (o.x.n == null ? '—' : String(o.x.n)) + '</td>' +
        '<td>' + escapeHtml(o.x.source || '—') + '</td>' +
        '</tr>';
    }).join('');
    tbody.innerHTML = html || '<tr><td colspan="6">No matches</td></tr>';
    if (resolved.length > PAGE) {
      count.textContent += ' (showing first ' + PAGE + ' — refine search)';
    }
  }

  function escapeHtml(s) {
    return String(s).replace(/[&<>"']/g, function (c) {
      return ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[c];
    });
  }

  function onSearch() {
    render(L.filterRows(allRows, search.value));
  }

  fetch('./prices.json')
    .then(function (res) { return res.json(); })
    .then(function (data) {
      allRows = data.rows || [];
      const m = data.meta || {};
      metaLine.textContent =
        'Data accessed ' + (m.accessed || '—') +
        '. Lettings ' + (m.let_date_min || '—') + ' → ' + (m.let_date_max || '—') +
        '. Houston first; statewide fallback labeled.';
      onSearch();
    })
    .catch(function (err) {
      metaLine.textContent = 'Could not load prices.json';
      console.error(err);
    });

  search.addEventListener('input', onSearch);
})();
```
